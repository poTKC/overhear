# ライブラリ（端末での選曲）実装計画

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 利用者の Spotify のライブラリと検索を端末で階層的にたどり、選んだ曲から再生できるようにする。あわせて、
キューの曲へのジャンプを Web API で一発にする。

トークンの管理、ログインのコールバックの照合、階層の保存をまとめ、ホストテストで確かめる。画面は
`LibraryActivity`（階層の一覧）と `LoginActivity`（QR でのログイン）を新しく作る。
設計は [2026-10-27-library-design.md](2026-10-27-library-design.md) を参照する。

**Tech Stack:** Java（API 18、Gradle なし、無名クラスで書きラムダは使わない、org.json は API 18 にある部分だけ）、
Go 1.27.3（helper）、JUnit 4（`test/run.sh`）、Nayuki qrcodegen（QR）。

```go
// replyServer answers every request with status, an optional Retry-After and
// body, and records the last request for the test to inspect.
func replyServer(t *testing.T, status int, retryAfter, body string) (*httptest.Server, *http.Request, *[]byte) {
	t.Helper()
	var last http.Request
	var lastBody []byte
	srv := httptest.NewTLSServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		last = *r
		lastBody, _ = io.ReadAll(r.Body)
		if retryAfter != "" {
			w.Header().Set("Retry-After", retryAfter)
		}
		w.WriteHeader(status)
		io.WriteString(w, body)
	}))
	t.Cleanup(srv.Close)
	return srv, &last, &lastBody
}

func caFor(t *testing.T, srv *httptest.Server) string {
	t.Helper()
	ca := filepath.Join(t.TempDir(), "ca.pem")
	pemBytes := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: srv.Certificate().Raw})
	if err := os.WriteFile(ca, pemBytes, 0o644); err != nil {
		t.Fatal(err)
	}
	return ca
}
```

```go
// postToken sends the form read from stdin to the token endpoint.
func postToken(ctx context.Context, client *http.Client, stdin io.Reader, stdout io.Writer) error {
	form, err := io.ReadAll(io.LimitReader(stdin, 64<<10))
	if err != nil {
		return err
	}
	req, err := http.NewRequestWithContext(ctx, "POST", accountsBase+"/api/token", bytes.NewReader(form))
	if err != nil {
		return err
	}
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	return doReply(client, req, stdout)
}

// doReply runs req and writes "status N", an optional "retry-after V", a
// blank line and the body (at most maxReplyBody bytes) to stdout.
func doReply(client *http.Client, req *http.Request, stdout io.Writer) error {
	req.Header.Set("User-Agent", userAgent)
	resp, err := client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	body, err := io.ReadAll(io.LimitReader(resp.Body, maxReplyBody+1))
	if err != nil {
		return err
	}
	if len(body) > maxReplyBody {
		return fmt.Errorf("reply larger than %d bytes", maxReplyBody)
	}
	fmt.Fprintf(stdout, "status %d\n", resp.StatusCode)
	if ra := resp.Header.Get("Retry-After"); ra != "" {
		fmt.Fprintf(stdout, "retry-after %s\n", strings.TrimSpace(ra))
	}
	fmt.Fprint(stdout, "\n")
	_, err = stdout.Write(body)
	return err
}
```

```go
// checkAPIPath accepts an absolute path (with an optional query) below
// apiBase: no scheme or host, no dot segments, no fragment.
func checkAPIPath(p string) error {
	if !strings.HasPrefix(p, "/") || strings.HasPrefix(p, "//") || strings.ContainsAny(p, "#\\") {
		return fmt.Errorf("bad API path %q", p)
	}
	path := p
	if i := strings.IndexByte(path, '?'); i >= 0 {
		path = path[:i]
	}
	for _, seg := range strings.Split(path, "/") {
		if seg == "." || seg == ".." {
			return fmt.Errorf("bad API path %q", p)
		}
	}
	return nil
}

// callAPI sends METHOD apiBase+path with the bearer token from
// $OVERHEAR_TOKEN and the body read from stdin (JSON when not empty).
func callAPI(ctx context.Context, client *http.Client, method, path string, stdin io.Reader, stdout io.Writer) error {
	body, err := io.ReadAll(io.LimitReader(stdin, 64<<10))
	if err != nil {
		return err
	}
	req, err := http.NewRequestWithContext(ctx, method, apiBase+path, bytes.NewReader(body))
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer "+os.Getenv("OVERHEAR_TOKEN"))
	if len(body) > 0 {
		req.Header.Set("Content-Type", "application/json")
	}
	return doReply(client, req, stdout)
}
```

```java
package com.potkc.overhear.spotify;

import java.io.IOException;
import java.io.UnsupportedEncodingException;

/**
 * One reply of overhear-fetch's token and api subcommands: "status N", an
 * optional "retry-after V", a blank line, then the body as sent. Pure Java.
 */
public final class HelperReply {
    public final int status;
    /** Seconds from Retry-After, or -1 when absent or not a number of seconds. */
    public final int retryAfterSeconds;
    public final String body;

    public HelperReply(int status, int retryAfterSeconds, String body) {
        this.status = status;
        this.retryAfterSeconds = retryAfterSeconds;
        this.body = body;
    }

    public static HelperReply parse(byte[] out) throws IOException {
        String s;
        try {
            s = new String(out, "UTF-8");
        } catch (UnsupportedEncodingException e) {
            throw new AssertionError(e);
        }
        int end = s.indexOf("\n\n");
        if (end < 0) throw new IOException("no header end in helper reply");
        int status = -1, retry = -1;
        for (String line : s.substring(0, end).split("\n")) {
            if (line.startsWith("status ")) status = parseInt(line.substring(7), -1);
            else if (line.startsWith("retry-after ")) retry = parseInt(line.substring(12), -1);
        }
        if (status < 100) throw new IOException("no status in helper reply");
        return new HelperReply(status, retry, s.substring(end + 2));
    }

    private static int parseInt(String s, int fallback) {
        try {
            return Integer.parseInt(s.trim());
        } catch (NumberFormatException e) {
            return fallback;
        }
    }
}
```

```java
package com.potkc.overhear.spotify;

import java.io.UnsupportedEncodingException;
import java.net.URLEncoder;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;

/**
 * One login attempt's PKCE values (RFC 7636, S256) and the authorize URL.
 * The state is "<device IPv4>:<port>~<nonce>": the relay page on GitHub
 * Pages sends the browser back to that address on the LAN. Pure Java.
 */
public final class Pkce {
    public final String verifier;
    public final String challenge;
    public final String nonce;

    private Pkce(String verifier, String nonce) {
        this.verifier = verifier;
        this.challenge = challengeOf(verifier);
        this.nonce = nonce;
    }

    public static Pkce create(SecureRandom random) {
        return new Pkce(base64Url(bytes(random, 48)), base64Url(bytes(random, 32)));
    }

    public static String state(String ip, int port, String nonce) {
        return ip + ":" + port + "~" + nonce;
    }

    public static String authorizeUrl(String clientId, String redirectUri, String scopes, String challenge,
            String state) {
        return "https://accounts.spotify.com/authorize?client_id=" + enc(clientId)
                + "&response_type=code&redirect_uri=" + enc(redirectUri)
                + "&code_challenge_method=S256&code_challenge=" + enc(challenge)
                + "&scope=" + enc(scopes) + "&state=" + enc(state);
    }

    static String challengeOf(String verifier) {
        try {
            return base64Url(MessageDigest.getInstance("SHA-256").digest(verifier.getBytes("US-ASCII")));
        } catch (NoSuchAlgorithmException e) {
            throw new AssertionError(e);
        } catch (UnsupportedEncodingException e) {
            throw new AssertionError(e);
        }
    }

    static String enc(String s) {
        try {
            return URLEncoder.encode(s, "UTF-8");
        } catch (UnsupportedEncodingException e) {
            throw new AssertionError(e);
        }
    }

    private static byte[] bytes(SecureRandom random, int n) {
        byte[] b = new byte[n];
        random.nextBytes(b);
        return b;
    }

    private static final char[] B64 =
            "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_".toCharArray();

    /** Base64url without padding (java.util.Base64 is API 26). */
    static String base64Url(byte[] b) {
        StringBuilder sb = new StringBuilder((b.length * 4 + 2) / 3);
        for (int i = 0; i < b.length; i += 3) {
            int n = (b[i] & 0xff) << 16;
            if (i + 1 < b.length) n |= (b[i + 1] & 0xff) << 8;
            if (i + 2 < b.length) n |= b[i + 2] & 0xff;
            sb.append(B64[n >> 18 & 63]).append(B64[n >> 12 & 63]);
            if (i + 1 < b.length) sb.append(B64[n >> 6 & 63]);
            if (i + 2 < b.length) sb.append(B64[n & 63]);
        }
        return sb.toString();
    }
}
```

```java
package com.potkc.overhear.spotify;

import java.io.UnsupportedEncodingException;
import java.net.URLDecoder;

/**
 * Classifies the request line the login listener received: the code, an
 * error Spotify sent back (e.g. access_denied), or anything else, which is
 * rejected. Only "GET /callback?..." with exactly the expected state counts.
 * Pure Java.
 */
public final class OAuthCallback {
    public enum Kind { CODE, ERROR, REJECT }

    public final Kind kind;
    /** The code (CODE) or the error (ERROR); null for REJECT. */
    public final String value;

    private OAuthCallback(Kind kind, String value) {
        this.kind = kind;
        this.value = value;
    }

    private static final OAuthCallback REJECTED = new OAuthCallback(Kind.REJECT, null);

    public static OAuthCallback parse(String requestLine, String expectedState) {
        if (requestLine == null) return REJECTED;
        String[] parts = requestLine.split(" ");
        if (parts.length != 3 || !"GET".equals(parts[0]) || !parts[2].startsWith("HTTP/")) return REJECTED;
        String target = parts[1];
        if (!target.startsWith("/callback?")) return REJECTED;
        String code = null, error = null, state = null;
        for (String pair : target.substring("/callback?".length()).split("&")) {
            int eq = pair.indexOf('=');
            if (eq <= 0) continue;
            String k = pair.substring(0, eq), v = dec(pair.substring(eq + 1));
            if (v == null) return REJECTED;
            if ("code".equals(k)) code = v;
            else if ("error".equals(k)) error = v;
            else if ("state".equals(k)) state = v;
        }
        if (state == null || !state.equals(expectedState)) return REJECTED;
        if (error != null) return new OAuthCallback(Kind.ERROR, error);
        if (code == null || code.isEmpty()) return REJECTED;
        return new OAuthCallback(Kind.CODE, code);
    }

    private static String dec(String s) {
        try {
            return URLDecoder.decode(s, "UTF-8");
        } catch (UnsupportedEncodingException e) {
            throw new AssertionError(e);
        } catch (IllegalArgumentException e) {
            return null; // bad % escape
        }
    }
}
```

```java
 * and the access token (memory only), refreshed a minute before it expires
 * or when the API answered 401. Spotify refusing the refresh token
 * (invalid_grant: revoked, or six months after the first login) clears the
 * store. Never logs tokens. Thread-safe (synchronized); calls block.
 * Pure Java.
 */
public final class SpotifyAuth {
    public interface TokenStore {
        /** The refresh token or null. */
        String refreshToken();

        void saveRefreshToken(String token);

        void clear();
    }

    public interface Clock {
        long nowMs();
    }

    private static final long EARLY_MS = 60 * 1000;

    private final ApiTransport transport;
    private final TokenStore store;
    private final Clock clock;
    private final String clientId;
    private String access; // guarded by this
    private long expiresAtMs; // guarded by this

    public SpotifyAuth(ApiTransport transport, TokenStore store, Clock clock, String clientId) {
        this.transport = transport;
        this.store = store;
        this.clock = clock;
        this.clientId = clientId;
    }

    public synchronized boolean isLoggedIn() {
        return store.refreshToken() != null;
    }

    /** Trades a login's code for tokens and keeps the refresh token. */
    public synchronized void exchangeCode(String code, String verifier, String redirectUri) throws IOException {
        tokenRequest("grant_type=authorization_code&code=" + Pkce.enc(code) + "&redirect_uri="
                + Pkce.enc(redirectUri) + "&client_id=" + Pkce.enc(clientId) + "&code_verifier="
                + Pkce.enc(verifier));
    }

    /** A valid access token, refreshing it when needed. */
    public synchronized String accessToken() throws IOException {
        if (access != null && clock.nowMs() < expiresAtMs - EARLY_MS) return access;
        String refresh = store.refreshToken();
        if (refresh == null) throw loginRequired("not logged in");
        tokenRequest("grant_type=refresh_token&refresh_token=" + Pkce.enc(refresh) + "&client_id="
                + Pkce.enc(clientId));
        return access;
    }

    /** Forgets the access token (after a 401) so the next call refreshes. */
    public synchronized void invalidate() {
        access = null;
    }

    public synchronized void logOut() {
        access = null;
        store.clear();
    }

    private void tokenRequest(String form) throws IOException { // under this
        HelperReply r = transport.token(form);
        JSONObject j = parse(r.body);
        if (r.status == 400 && "invalid_grant".equals(j.optString("error"))) {
            logOut();
            throw loginRequired("login expired");
        }
        if (r.status != 200) throw SpotifyApi.httpError("token", r);
        String a = j.optString("access_token", "");
        if (a.isEmpty()) throw new IOException("token reply without access_token");
        access = a;
        expiresAtMs = clock.nowMs() + j.optLong("expires_in", 3600) * 1000;
        String newRefresh = j.optString("refresh_token", "");
        if (!newRefresh.isEmpty()) store.saveRefreshToken(newRefresh);
    }

    private static JSONObject parse(String body) {
        try {
            return new JSONObject(body);
        } catch (JSONException e) {
            return new JSONObject();
        }
    }

    private static SpotifyException loginRequired(String why) {
        return new SpotifyException(SpotifyException.Reason.LOGIN_REQUIRED, 0, -1, why);
    }
}
```

```java
// SpotifyApi.java (the base; Task 10 adds the endpoints)
package com.potkc.overhear.spotify;

import java.io.IOException;

import org.json.JSONException;
import org.json.JSONObject;

/**
 * The Spotify Web API calls Overhear makes, through an ApiTransport. Each
 * call blocks (run it off the main thread). A 401 refreshes the access token
 * and retries once. Pure Java.
 */
public final class SpotifyApi {
    private final ApiTransport transport;
    private final SpotifyAuth auth;

    public SpotifyApi(ApiTransport transport, SpotifyAuth auth) {
        this.transport = transport;
        this.auth = auth;
    }

    public SpotifyAuth auth() {
        return auth;
    }

    JSONObject getJson(String path) throws IOException {
        HelperReply r = call("GET", path, "");
        if (r.status != 200) throw httpError("GET " + stripQuery(path), r);
        try {
            return new JSONObject(r.body);
        } catch (JSONException e) {
            throw new IOException("bad JSON from " + stripQuery(path));
        }
    }

    /** PUT with a JSON body; any 2xx is success. */
    void put(String path, String body) throws IOException {
        HelperReply r = call("PUT", path, body);
        if (r.status / 100 != 2) throw httpError("PUT " + stripQuery(path), r);
    }

    private HelperReply call(String method, String path, String body) throws IOException {
        HelperReply r = transport.api(method, path, auth.accessToken(), body);
        if (r.status == 401) {
            auth.invalidate();
            r = transport.api(method, path, auth.accessToken(), body);
        }
        return r;
    }

    static SpotifyException httpError(String what, HelperReply r) {
        if (r.status == 429) {
            return new SpotifyException(SpotifyException.Reason.RATE_LIMITED, 429, r.retryAfterSeconds,
                    what + ": rate limited");
        }
        return new SpotifyException(SpotifyException.Reason.HTTP, r.status, -1, what + ": HTTP " + r.status);
    }

    private static String stripQuery(String path) {
        int q = path.indexOf('?');
        return q < 0 ? path : path.substring(0, q);
    }
}
```

```java
import com.potkc.overhear.soloist.PlayerState;

/**
 * Decides how to reach a tapped queue entry: one Web API play of the context
 * from that track, or QueueJump's skip_next steps. The Web API works only
 * for tracks of the playing context (not queued or autoplay ones) in a
 * context Spotify takes an offset for (albums, playlists, liked songs), and
 * picks the first occurrence of a URI, so a duplicate ahead of the target
 * rules it out. Pure Java.
 */
final class QueueRoute {
    private QueueRoute() {}

    static boolean useWebApi(List<PlayerState.QueueItem> queue, int index, String contextUri, boolean shuffle,
            boolean loggedIn) {
        if (!loggedIn || shuffle || contextUri == null || index < 0 || index >= queue.size()) return false;
        if (!(contextUri.startsWith("spotify:album:") || contextUri.startsWith("spotify:playlist:")
                || (contextUri.startsWith("spotify:user:") && contextUri.endsWith(":collection")))) {
            return false;
        }
        PlayerState.QueueItem target = queue.get(index);
        if (!"context".equals(target.source) || target.uri == null) return false;
        for (int i = 0; i < index; i++) {
            PlayerState.QueueItem before = queue.get(i);
            if (target.uri.equals(before.uri) || !"context".equals(before.source)) return false;
        }
        return true;
    }
}
```

**Step 2: LoginActivity の流れ**

1. `onCreate`: Wi-Fi の IP を `WifiManager.getConnectionInfo().getIpAddress()` から取る（リトルエンディアンの int なので
   変換に注意）。0 なら「Wi-Fi につないでください」と表示して終わる。
2. `Pkce.create(random)` で値を作り、state を `Pkce.state(ip, 8889, nonce)` にする。`random` は `new SecureRandom()` に、
   `/dev/urandom` から読んだ 64 バイトで `setSeed` したもの（Android 4.1〜4.3 には、Zygote から fork したアプリで
   `SecureRandom` の種が弱くなる既知の不具合があるため。この ROM に修正が入っているかは分からない）。`/dev/urandom` が
   読めなければログインを始めずにエラーを表示する。
3. 待ち受けスレッドを起動する（`ServerSocket`、ポート 8889、`setSoTimeout(1000)` でループし、5 分か `onDestroy` で終える）。
   1 つの接続ごとに:
   - 1 行目（最大 8 KiB）を読み、`OAuthCallback.parse(line, state)` に渡す。
   - REJECT: `HTTP/1.1 400 Bad Request` を返して閉じる（ヘッダーまで読み飛ばしてから返す）。
   - ERROR: 「キャンセルされました」の HTML を返し、画面に「キャンセルされました。もう一度読み取れます」と出す
     （待ち受けは続ける）。
   - CODE: `Spotify.get(this).run(...)` で `auth().exchangeCode(code, verifier, Spotify.REDIRECT_URI)` と `api().me()` を行う。
     **交換が終わるまで応答を待たせ**、成功なら「ログインしました。このページは閉じてかまいません」、失敗なら
     「ログインできませんでした。端末の画面を確認してください」の HTML を返す。成功したら待ち受けを止め、
     画面に「ログインしました: <ユーザー ID>」と出して 2 秒後に `finish()`。
   - 返す HTML には `Referrer-Policy: no-referrer`、`Content-Type: text/html; charset=utf-8`、
     `Connection: close` を付ける。
4. QR に `Pkce.authorizeUrl(Spotify.CLIENT_ID, Spotify.REDIRECT_URI, Spotify.SCOPES, challenge, state)` を出す。
5. 5 分たったら「時間切れです。開き直してください」と出し、QR を隠す。

| SEARCH | `search(query)`。確定キーで実行し、結果は「曲」「アルバム」「アーティスト」「プレイリスト」の見出し行で区切る | 曲は `play(device, item.contextUri, item.uri)`（そのアルバムのその曲から）、ほかは各ページへ |

- 最後の行が表示されたら（`OnScrollListener`）、`next` があれば次を読み込む。読み込み中は最後に「読み込み中…」の行。
- 端末名は `new Prefs(this).getDeviceName()`。

**Step 4: 再生の指示と例外時の扱い**

- 再生は `Spotify.run(api.play(...))`。成功したら `finish()` して再生画面に戻る。
- 失敗したら（`SpotifyException` か `IOException`）、サービスの `SoloistClient` に `play(uri)` を送る（曲一覧の曲なら
  コンテキストの URI、コンテキストがなければ曲の URI）。トーストで「最初から再生します」と出し、`finish()`。
  サービスへのアクセスは `PlayerActivity` と同じく `bindService(..., 0)` で `LocalBinder` を取る。
- 一覧の取得に失敗したら、一覧の位置に次を出す。
  - `LOGIN_REQUIRED`: 「Spotify のログインが切れました — タップしてログイン」（タップで LoginActivity）
  - `RATE_LIMITED`: 「混み合っています。しばらくしてから ↻ を押してください」
  - それ以外: 「読み込めませんでした — タップで再試行」
- ジャケットは `Artwork.get(this)` の ArtworkLoader で `item.coverUrl` を読む（キューの行と同じ使い方）。

**Step 5: マニフェストとビルド**

`LibraryActivity` を `exported="false"` で足す。`SKIP_RUNTIME=1 ./build.sh` でビルドする。

**Step 6: 実機で確かめる**

各ページの表示とページ送り（プレイリストは 83 件あるので 2 ページ目まで）、検索、曲をタップしてその曲から再生、
「↻」、戻るキー、前回の階層からの再開。

### Task 19: ほかの人のプレイリストを開く

**Files:**
- Modify: `app/src/com/potkc/overhear/LibraryActivity.java`

**Step 1: 開いたときの流れ**

1. OTHERS_PLAYLIST のページを積み、「再生を始めています…」と表示する。
2. `Spotify.run(api.play(device, playlistUri, null))`。失敗したら `SoloistClient.play(playlistUri)`。
3. サービスの `SoloistClient` のリスナーで状態を受け取り、`state.getContextUri()` がそのプレイリストになったら
   `client.requestQueue(SoloistClient.FULL_QUEUE_LIMIT)` を 1 回だけ送る。
4. コンテキストがそのプレイリストになって最初に見えた曲の URI を `firstTrackUri` として記録する。状態が更新されるたびに
   `QueueTracks.fromState(state, playlistUri, firstTrackUri)` を呼び、null でなく件数が前回より多ければ一覧を更新する
   （あとから届く通常の 10 件の `queue_changed` で一覧が縮まないようにするため、**最初に取れた最大の一覧を保持する**。
   曲をタップして先頭の曲から離れると `fromState` は null を返すので、一覧は回転せずにそのまま残る）。
5. シャッフルがオンなら、一覧の先頭に「シャッフル順です」の行を出す。
6. 曲をタップしたら `play(device, playlistUri, item.uri)`。この画面だけは成功しても `finish()` しない
   （続けて選び直せるように）。

**Step 2: 実機で確かめる**

途中の曲をタップするとその曲になること。シャッフルがオンのときの表示。
